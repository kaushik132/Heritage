<?php

namespace App\Admin\Controllers;

use OpenAdmin\Admin\Controllers\AdminController;
use OpenAdmin\Admin\Form;
use OpenAdmin\Admin\Grid;
use OpenAdmin\Admin\Show;
use \App\Models\Product;
use \App\Models\ProductCategory;
use Illuminate\Support\Str;


class ProductController extends AdminController
{
    /**
     * Title for current resource.
     *
     * @var string
     */
    protected $title = 'Product';

    /**
     * Make a grid builder.
     *
     * @return Grid
     */
    protected function grid()
    {
        $grid = new Grid(new Product());

        $grid->column('id', __('Id'));
        $grid->column('name', __('Name'));
        $grid->column('slug', __('Slug'));
        $grid->column('category', __('Category'));
        $grid->column('short_description', __('Short description'));
        $grid->column('description', __('Description'));
        $grid->column('price', __('Price'));
        $grid->column('moq', __('Moq'));
        $grid->column('stock', __('Stock'));
        $grid->column('in_stock', __('In stock'));
        $grid->column('weight', __('Weight'));
        $grid->column('width', __('Width'));
        $grid->column('weave', __('Weave'));
        $grid->column('origin', __('Origin'));
        $grid->column('composition', __('Composition'));
        $grid->column('care_instructions', __('Care instructions'));
        $grid->column('india_shipping', __('India shipping'));
        $grid->column('international_shipping', __('International shipping'));
        $grid->column('colors', __('Colors'))->display(function ($colors) {

            if (is_string($colors)) {
                $colors = json_decode($colors, true);
            }

            return is_array($colors) ? implode(', ', $colors) : $colors;
        });
        $grid->column('rating', __('Rating'));
        $grid->column('review_count', __('Review count'));
        $grid->column('is_featured', __('Is featured'));
        $grid->column('is_active', __('Is active'));
        $grid->column('created_at', __('Created at'));
        $grid->column('updated_at', __('Updated at'));
        $grid->column('type', __('Type'));

        return $grid;
    }

    /**
     * Make a show builder.
     *
     * @param mixed $id
     * @return Show
     */
    protected function detail($id)
    {
        $show = new Show(Product::findOrFail($id));

        $show->field('id', __('Id'));
        $show->field('name', __('Name'));
        $show->field('slug', __('Slug'));
        $show->field('category', __('Category'));
        $show->field('short_description', __('Short description'));
        $show->field('description', __('Description'));
        $show->field('price', __('Price'));
        $show->field('moq', __('Moq'));
        $show->field('stock', __('Stock'));
        $show->field('in_stock', __('In stock'));
        $show->field('weight', __('Weight'));
        $show->field('width', __('Width'));
        $show->field('weave', __('Weave'));
        $show->field('origin', __('Origin'));
        $show->field('composition', __('Composition'));
        $show->field('care_instructions', __('Care instructions'));
        $show->field('india_shipping', __('India shipping'));
        $show->field('international_shipping', __('International shipping'));
        $show->field('colors', __('Colors'))->as(function ($colors) {

            if (is_string($colors)) {
                $colors = json_decode($colors, true);
            }

            return is_array($colors) ? implode(', ', $colors) : $colors;
        });
        $show->field('rating', __('Rating'));
        $show->field('review_count', __('Review count'));
        $show->field('is_featured', __('Is featured'));
        $show->field('is_active', __('Is active'));
        $show->field('created_at', __('Created at'));
        $show->field('updated_at', __('Updated at'));
        $show->field('type', __('Type'));

        return $show;
    }

    /**
     * Make a form builder.
     *
     * @return Form
     */
    protected function form()
    {
        $form = new Form(new Product());
        $form->tab('Product Info', function ($form) {
            $form->text('name', __('Name'));

            $form->hidden('slug', __('Slug'));
            $form->saving(function (Form $form) {
                $form->slug = Str::slug($form->name);
            });

            $form->select('category', __('Category Name'))->options(ProductCategory::pluck('name', 'id'));

            $form->textarea('short_description', __('Short description'));
            $form->textarea('description', __('Description'));
            $form->decimal('price', __('Price'));
            $form->number('moq', __('Moq'))->default(50);
            $form->number('stock', __('Stock'));
            $form->switch('in_stock', __('In stock'))->default(1);
            $form->number('weight', __('Weight'));
            $form->text('width', __('Width'));
            $form->text('weave', __('Weave'));
            $form->text('origin', __('Origin'));
            $form->text('composition', __('Composition'));
            $form->text('care_instructions', __('Care instructions'));
            $form->text('india_shipping', __('India shipping'));
            $form->text('international_shipping', __('International shipping'));
            $form->tags('colors', __('Colors'));

            $form->saved(function (Form $form) {
                $product = $form->model();
                $raw = \DB::table('products')->where('id', $product->id)->value('colors');

                // Agar already valid JSON array hai to skip karo
                if (!$raw) return;

                $decoded = json_decode($raw, true);

                // Already valid array hai — kuch karne ki zaroorat nahi
                if (is_array($decoded)) return;

                // String format hai — fix karo
                if (is_string($raw) && !str_starts_with(trim($raw), '[')) {
                    $cleaned = trim($raw, '"\\');
                    $colors = array_values(array_filter(
                        array_map(function ($color) {
                            return trim($color, '" \\');
                        }, explode(',', $cleaned)),
                        fn($c) => !empty($c) // empty strings hata do
                    ));

                    \DB::table('products')
                        ->where('id', $product->id)
                        ->update(['colors' => json_encode($colors)]);
                }
            });
            $form->decimal('rating', __('Rating'))->default(0.0);
            $form->number('review_count', __('Review count'));
            $form->switch('is_featured', __('Is featured'));
            $form->switch('is_active', __('Is active'))->default(1);
            $form->select('type', __('Type'))->options(['retailer' => 'retailer', 'wholeseller' => 'wholeseller']);


                 $form->hasMany('images', 'Product Images', function (Form\NestedForm $form) {
            $form->number('sort_order', __('Order'))->default(0);
            $form->text('color', __('Color'))->placeholder('Green, Red, Blue...');
            $form->image('image_path', __('Image'));
            $form->switch('is_primary', __('Is Primary'));
        });
        });




        return $form;
    }
}
